              if (x > right - border) edge |= Align.right;
              if (y < bottom + border) edge |= Align.bottom;
              if (edge != 0) border += 25;
              if (x < left + border) edge |= Align.left;
              if (x > right - border) edge |= Align.right;
              if (y < bottom + border) edge |= Align.bottom;
