public InputStream read() {
    if (type == FileType.Classpath
        || (type == FileType.Internal && !file().exists())
        || (type == FileType.Local && !file().exists())) {
      File classpathFile = file();
      if (classpathFile == null)
        throw new GdxRuntimeException("File not found: " + file + " (" + type + ")");
      InputStream input =
          FileHandle.class.getResourceAsStream("/" + classpathFile.getPath().replace('\\', '/'));
      if (input == null)
        throw new GdxRuntimeException("File not found: " + file + " (" + type + ")");
      return input;
    }
    try {
      return new FileInputStream(file());
    } catch (Exception ex) {
      if (file().isDirectory())
        throw new GdxRuntimeException(
            "Cannot open a stream to a directory: " + file + " (" + type + ")", ex);
      throw new GdxRuntimeException("Error reading file: " + file + " (" + type + ")", ex);
    }
}
