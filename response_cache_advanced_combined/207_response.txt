        Pixmap pixmap = texData.consumePixmap();

          if (texData.disposePixmap()) {
            pixmap.dispose();
          }
          pixmap = temp;
          disposePixmap = true;
        }

            0,
            0,

            1,

            pixmap.getGLType(),
            pixmap.getPixels());

      }
