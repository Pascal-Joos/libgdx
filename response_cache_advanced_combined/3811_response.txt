public int getKeyFrameIndex(float stateTime) {
    if (keyFrames == null || keyFrames.length == 0) return 0;
    if (Nullability.castToNonnull(keyFrames).length == 1) return 0;

    Object[] nonNullKeyFrames = Nullability.castToNonnull(keyFrames);

    int frameNumber = (int) (stateTime / frameDuration);
    switch (playMode) {
      case NORMAL:
        frameNumber = Math.min(nonNullKeyFrames.length - 1, frameNumber);
        break;
      case LOOP:
        frameNumber = frameNumber % nonNullKeyFrames.length;
        break;
      case LOOP_PINGPONG:
        frameNumber = frameNumber % ((nonNullKeyFrames.length * 2) - 2);
        if (frameNumber >= nonNullKeyFrames.length)
          frameNumber =
              nonNullKeyFrames.length
                  - 2
                  - (frameNumber - nonNullKeyFrames.length);
        break;
      case LOOP_RANDOM:
        int lastFrameNumber = (int) ((lastStateTime) / frameDuration);
        if (lastFrameNumber != frameNumber) {
          frameNumber = MathUtils.random(nonNullKeyFrames.length - 1);
        } else {
          frameNumber = this.lastFrameNumber;
        }
        break;
      case REVERSED:
        frameNumber = Math.max(nonNullKeyFrames.length - frameNumber - 1, 0);
        break;
      case LOOP_REVERSED:
        frameNumber = frameNumber % nonNullKeyFrames.length;
        frameNumber = nonNullKeyFrames.length - frameNumber - 1;
        break;
    }

    lastFrameNumber = frameNumber;
    lastStateTime = stateTime;

    return frameNumber;
}
