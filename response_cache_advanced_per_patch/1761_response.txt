      if (!(e instanceof InputEvent)) return false;
      InputEvent event = (InputEvent) e;
  
      switch (event.getType()) {
        case touchDown:
          actor = event.getListenerActor();
          if (actor == null) return false;

                    event.getTarget(),
                    event.getPointer(),
                    event.getButton());
          return true;
        case touchUp:
          if (event.isTouchFocusCancel()) {
            detector.reset();
            return false;
          }
          this.event = event;
          actor = event.getListenerActor();
          if (actor == null) return false;

          return true;
        case touchDragged:

          if (actor == null) return false;

          return true;
      }
      return false;
}
